fn offsetMomentum(b, nbody, solar_mass) {
    let px = 0.0;
    let bi = b[0];
    let bim = bi.mass;
    px += bi.vx * bim;
    b[0].vx = -px / solar_mass;
}

fn pad_left(s, width, fill) {
    let out = s;
    while out.len() < width {
        out = fill + out;
    }
    return out;
}

fn unwrap(o) {
    let value = null;
    match o {
        Some(v) => { value = v; }
        None => { throw("called unwrap on a None option"); }
        _ => { value = null; }
    }
    return value;
}
